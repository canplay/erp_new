## 1. 项目架构概述

MyAI 前端采用 **Vue 3 + Quasar Framework** 技术栈，基于 TypeScript 开发，使用 Pinia 进行状态管理，Vue Router 进行路由管理。

```

│                    MyAI Admin Frontend                    │
│              (Vue 3 + Quasar + TypeScript)                │

                                         │
```

## 7. 路由系统

### 7.1 路由架构

项目使用 **Vue Router 5** 进行路由管理，采用动态路由加载和权限控制。

```typescript
// 路由配置示例
const routes: RouteRecordRaw[] = [
  {
    path: '/login',
    name: 'Login',
    component: () => import('@/pages/LoginPage.vue'),
    meta: { title: '登录', requiresAuth: false },
  },
  {
    path: '/',
    component: () => import('@/layouts/MainLayout.vue'),
    meta: { requiresAuth: true },
    children: [
      {
        path: '',
        name: 'Dashboard',
        component: () => import('@/pages/DashboardPage.vue'),
        meta: { title: '仪表盘', icon: 'dashboard' },
      },
      // ... 更多路由
    ],
  },
];
```

### 7.2 路由守卫

项目实现了完整的路由守卫链：

```typescript
// 1. 前置守卫 - 路由鉴权
router.beforeEach(async (to, _from, next) => {
  // 更新页面标题
  if (to.meta.title) {
    document.title = `${to.meta.title} - 管理后台`;
  }

  // 需要登录的页面
  if (to.meta.requiresAuth) {
    if (!authStore.isLoggedIn) {
      next({ path: '/login', query: { redirect: to.fullPath } });
      return;
    }

    // 初始化用户信息
    if (!authStore.userInfo) {
      await authStore.fetchUserInfo();
    }

    // 权限检查
    if (to.meta.permissions?.length > 0) {
      const hasPermission = to.meta.permissions.some(
        (perm: string) => authStore.hasPermission(perm)
      );
      if (!hasPermission) {
        next({ path: '/403' });
        return;
      }
    }
  }

  next();
});
```

### 7.3 路由预加载

项目实现了路由预加载机制，提升页面切换速度：

```typescript
// 预加载关键页面
const preloadRoutes = ['Dashboard', 'Profile', 'UserList'];

setTimeout(() => {
  preloadRoutes.forEach((routeName) => {
    const routes = router.getRoutes().filter((r) => r.name === routeName);
    routes.forEach((r) => {
      if (r.components?.default) {
        const component = r.components.default as { preload?: () => void };
        if (typeof component.preload === 'function') {
          component.preload();
        }
      }
    });
  });
}, 3000); // 3秒后开始预加载
```

## 8. 状态管理

### 8.1 Pinia Store 架构

项目使用 **Pinia** 进行状态管理，采用模块化设计：

```typescript
// stores/auth.ts - 认证状态
export const useAuthStore = defineStore('auth', {
  state: () => ({
    token: localStorage.getItem('token') || '',
    refreshToken: localStorage.getItem('refreshToken') || '',
    userInfo: null as UserInfo | null,
    isLoggedIn: false,
  }),
  getters: {
    hasPermission: (state) => (perm: string) => {
      return state.userInfo?.permissions?.includes(perm) ?? false;
    },
  },
  actions: {
    async login(username: string, password: string) {
      // 登录逻辑
    },
    async fetchUserInfo() {
      // 获取用户信息
    },
    logout() {
      // 登出逻辑
    },
  },
});
```

```typescript
  'role:list',
  'role:read',
```

### 9.2 权限检查流程

```typescript
// 1. 路由守卫检查
if (to.meta.permissions?.length > 0) {
  const hasPermission = to.meta.permissions.some(
    (perm: string) => authStore.hasPermission(perm)
  );
  if (!hasPermission) {
    next({ path: '/403' });

// 2. 组件内权限检查
const { hasPermission } = useAuthStore();
if (!hasPermission('user:list')) {
  // 隐藏按钮或显示提示

```

## 11. Nginx 配置

### 11.1 Nginx 配置要点

项目使用 `nginx.conf` 进行 Nginx 配置，包含以下关键配置：

```nginx
# 日志输出到 stdout/stderr (适配容器)
access_log /dev/stdout;
error_log /dev/stderr;

# 健康检查端点
location = /health {
    return 200 'ok';
}
location = /ready {
    return 200 'ready';
}

# SPA 路由支持
location / {
    try_files $uri @spa;
}
location @spa {
    rewrite ^ /index.html last;
}

# API 代理到后端
location /api/ {
    proxy_pass http://api-gateway:8090;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    
    # 超时设置
    proxy_connect_timeout 60s;
    proxy_send_timeout 60s;
    proxy_read_timeout 120s;
    
    # CORS headers
    add_header Access-Control-Allow-Origin https://admin.100.100.100.100.example.com always;
    add_header Access-Control-Allow-Methods "GET, POST, PUT, DELETE, PATCH, OPTIONS" always;
    add_header Access-Control-Allow-Headers "Content-Type, Authorization, X-Requested-With" always;
}

# WebSocket 支持
location /ws {
    proxy_pass http://api-gateway:8090;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
}

# 静态资源缓存
location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
    expires 1y;
    add_header Cache-Control "public, immutable";
}

# 安全头
add_header X-Frame-Options "SAMEORIGIN" always;
add_header X-Content-Type-Options "nosniff" always;
add_header Content-Security-Policy "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'; img-src 'self' data:; font-src 'self' data:; connect-src 'self' wss://api.100.100.100.100.example.com:8901; frame-src 'self'; object-src 'none'; base-uri 'self'" always;
add_header X-XSS-Protection "1; mode=block" always;
```

## 13. 常见问题与排查

### 13.1 编译错误

**问题**: `Cannot find module '@myai/shared'`

```bash
# 解决方案: 重新安装依赖
cd Frontend
pnpm install --force
```

**问题**: `Vue Router 类型错误`

```bash
# 解决方案: 检查 TypeScript 版本
pnpm add -D typescript@6.0.3
```

### 13.2 运行时错误

**问题**: `API 请求失败: Network Error`

```bash
# 解决方案: 检查 API 地址配置
# .env.development:
VITE_API_BASE_URL=http://localhost:8090/api

# .env.production:
VITE_API_BASE_URL=https://api.100.100.100.100.example.com:8901/api
```

**问题**: `WebSocket 连接失败`

```bash
# 解决方案: 检查 WebSocket 地址配置
# .env.development:
VITE_WS_URL=ws://localhost:8090/ws

# .env.production:
VITE_WS_URL=wss://api.100.100.100.100.example.com:8901/ws
```

### 13.3 权限问题

**现象**: 登录后仍显示 403 页面

```bash
# 解决方案: 检查权限预加载
# auth.ts - 路由守卫
if (!permissionStore.isLoaded) {
  await permissionStore.fetchPermissions();
}
```

### 13.4 构建优化问题

**现象**: 构建体积过大

```bash
# 解决方案: 检查代码分割配置
# quasar.config.ts
manualChunks: (id) => {
  if (id.includes('node_modules/quasar')) return 'quasar';
  if (id.includes('node_modules/vue')) return 'vue';
  // ...
}
```
